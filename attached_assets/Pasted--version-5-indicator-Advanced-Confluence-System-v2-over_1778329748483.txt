//@version=5
indicator("Advanced Confluence System [v2]", overlay=true, max_lines_count=200, max_labels_count=200)

// ══════════════════════════════════════════════
// INPUTS
// ══════════════════════════════════════════════
grp1 = "Trend Settings"
emaLen      = input.int(200, "Trend EMA Length",              group=grp1)
adxLen      = input.int(14,  "ADX Length",                    group=grp1)
adxThresh   = input.int(20,  "ADX Min Strength (avoid chop)", group=grp1)

grp2 = "Oscillators"
rsiLen      = input.int(14,  "RSI Length",    group=grp2)
stochLen    = input.int(14,  "Stoch Length",  group=grp2)
rsiOS       = input.int(45,  "RSI Oversold",  group=grp2)

grp3 = "Support & Resistance"
atrLen      = input.int(14,   "ATR Length",        group=grp3)
atrMult     = input.float(1.5, "ATR Zone Multiplier", group=grp3, step=0.1)
pivotLB     = input.int(10,   "Pivot Lookback",    group=grp3)

grp4 = "Signal Settings"
minPoints   = input.int(3,    "Min Score to Signal (max 6)", group=grp4, minval=2, maxval=6)
showRR      = input.bool(true, "Show Risk/Reward Levels",    group=grp4)
rrRatio     = input.float(2.0, "Risk:Reward Ratio",          group=grp4, step=0.5)
showDash    = input.bool(true, "Show Dashboard",             group=grp4)

// ══════════════════════════════════════════════
// CORE CALCULATIONS
// ══════════════════════════════════════════════

// 1. Trend
ema200   = ta.ema(close, emaLen)
aboveEma = close > ema200
belowEma = close < ema200

// 2. ADX
[diPlus, diMinus, adx] = ta.dmi(adxLen, adxLen)
trendStrong = adx >= adxThresh

// 3. RSI + Stochastic
rsi    = ta.rsi(close, rsiLen)
stochK = ta.sma(ta.stoch(close, high, low, stochLen), 3)

// 4. MACD
[macdLine, signalLine, hist] = ta.macd(close, 12, 26, 9)
macdTurningUp   = hist > hist[1] and hist[1] <= hist[2]
macdTurningDown = hist < hist[1] and hist[1] >= hist[2]
macdBull        = hist > hist[1] and hist > 0
macdBear        = hist < hist[1] and hist < 0

// 5. ATR-based Dynamic S/R
atr     = ta.atr(atrLen)
pivotHi = ta.pivothigh(high, pivotLB, pivotLB)
pivotLo = ta.pivotlow(low,  pivotLB, pivotLB)

var float lastResistance = na
var float lastSupport    = na
if not na(pivotHi)
    lastResistance := pivotHi
if not na(pivotLo)
    lastSupport := pivotLo

nearRes = not na(lastResistance[1]) and high >= lastResistance[1] - atr * atrMult and high <= lastResistance[1] + atr * atrMult
nearSup = not na(lastSupport[1])    and low  <= lastSupport[1]    + atr * atrMult and low  >= lastSupport[1]    - atr * atrMult

// 6. Volume confirmation
volSma  = ta.sma(volume, 20)
highVol = volume > volSma * 1.2

// 7. Candle patterns
bullEngulf = close > open and close > open[1] and open < close[1] and close > high[1]  * 0.998
bearEngulf = close < open and close < open[1] and open > close[1] and close < low[1]   * 1.002
bullPin    = (high - math.max(open, close)) < (math.min(open, close) - low) * 0.3 and (math.min(open, close) - low) > atr * 0.6
bearPin    = (math.min(open, close) - low) < (high - math.max(open, close)) * 0.3 and (high - math.max(open, close)) > atr * 0.6

// ══════════════════════════════════════════════
// SCORING ENGINE (FIXED — removed explicit int declaration)
// ══════════════════════════════════════════════

// ── BUY SCORE ──
buyScore = 0
if aboveEma
    buyScore += 1
if trendStrong
    buyScore += 1
if rsi < rsiOS or stochK < 30
    buyScore += 1
if macdTurningUp or macdBull
    buyScore += 1
if nearSup
    buyScore += 1
if highVol
    buyScore += 1

bullCandle = (close > open) or bullEngulf or bullPin
buySignal  = (buyScore >= minPoints) and bullCandle

// ── SELL SCORE ──
sellScore = 0
if belowEma
    sellScore += 1
if trendStrong
    sellScore += 1
if rsi > (100 - rsiOS) or stochK > 70
    sellScore += 1
if macdTurningDown or macdBear
    sellScore += 1
if nearRes
    sellScore += 1
if highVol
    sellScore += 1

bearCandle = (close < open) or bearEngulf or bearPin
sellSignal = (sellScore >= minPoints) and bearCandle

// ══════════════════════════════════════════════
// SIGNAL GRADE
// ══════════════════════════════════════════════
buyGrade  = buyScore  >= 5 ? "A" : buyScore  == 4 ? "B" : "C"
sellGrade = sellScore >= 5 ? "A" : sellScore == 4 ? "B" : "C"

// ══════════════════════════════════════════════
// VISUALS
// ══════════════════════════════════════════════
plot(ema200, "EMA 200", color=aboveEma ? color.new(color.aqua, 20) : color.new(color.orange, 20), linewidth=2)
plot(lastResistance, "Resistance", color=color.new(color.red,  60), style=plot.style_stepline, linewidth=1)
plot(lastSupport,    "Support",    color=color.new(color.lime, 60), style=plot.style_stepline, linewidth=1)

plotshape(buySignal,  style=shape.labelup,   location=location.belowbar,
     color=buyScore  >= 5 ? color.new(color.green, 0) : color.new(color.green, 40),
     textcolor=color.white, text="▲", size=size.small)

plotshape(sellSignal, style=shape.labeldown, location=location.abovebar,
     color=sellScore >= 5 ? color.new(color.red, 0)   : color.new(color.red,   40),
     textcolor=color.white, text="▼", size=size.small)

if buySignal
    label.new(bar_index, low - atr * 0.4,
         buyGrade + " (" + str.tostring(buyScore) + "/6)",
         style=label.style_none, textcolor=color.green, size=size.tiny)

if sellSignal
    label.new(bar_index, high + atr * 0.4,
         sellGrade + " (" + str.tostring(sellScore) + "/6)",
         style=label.style_none, textcolor=color.red, size=size.tiny)

// ══════════════════════════════════════════════
// RISK / REWARD LEVELS
// ══════════════════════════════════════════════
if showRR and buySignal
    stopLoss = low  - atr * 1.0
    tp1      = close + (close - stopLoss) * rrRatio
    tp2      = close + (close - stopLoss) * rrRatio * 1.5
    line.new(bar_index, stopLoss, bar_index + 8, stopLoss, color=color.new(color.red,  40), style=line.style_dashed)
    line.new(bar_index, tp1,      bar_index + 8, tp1,      color=color.new(color.lime, 40), style=line.style_dashed)
    line.new(bar_index, tp2,      bar_index + 8, tp2,      color=color.new(color.teal, 40), style=line.style_dotted)

if showRR and sellSignal
    stopLoss = high + atr * 1.0
    tp1      = close - (stopLoss - close) * rrRatio
    tp2      = close - (stopLoss - close) * rrRatio * 1.5
    line.new(bar_index, stopLoss, bar_index + 8, stopLoss, color=color.new(color.red,  40), style=line.style_dashed)
    line.new(bar_index, tp1,      bar_index + 8, tp1,      color=color.new(color.lime, 40), style=line.style_dashed)
    line.new(bar_index, tp2,      bar_index + 8, tp2,      color=color.new(color.teal, 40), style=line.style_dotted)

// ══════════════════════════════════════════════
// DASHBOARD TABLE (FIXED — proper var initialization)
// ══════════════════════════════════════════════
var table dash = table.new(position.top_right, 2, 9,
     bgcolor=color.new(color.black, 75), border_color=color.new(color.gray, 60), border_width=1)

if showDash and barstate.islast
    table.cell(dash, 0, 0, "Indicator",    text_color=color.silver, text_size=size.small)
    table.cell(dash, 1, 0, "Status",       text_color=color.silver, text_size=size.small)

    table.cell(dash, 0, 1, "Trend EMA",   text_color=color.white, text_size=size.small)
    table.cell(dash, 1, 1, aboveEma ? "▲ Bullish" : "▼ Bearish",
         text_color=aboveEma ? color.lime : color.red, text_size=size.small)

    table.cell(dash, 0, 2, "ADX Strength", text_color=color.white, text_size=size.small)
    table.cell(dash, 1, 2, str.tostring(math.round(adx)) + (trendStrong ? " ✓" : " (weak)"),
         text_color=trendStrong ? color.lime : color.orange, text_size=size.small)

    table.cell(dash, 0, 3, "RSI",         text_color=color.white, text_size=size.small)
    table.cell(dash, 1, 3, str.tostring(math.round(rsi, 1)),
         text_color=rsi < 35 ? color.lime : rsi > 65 ? color.red : color.silver, text_size=size.small)

    table.cell(dash, 0, 4, "Stoch %K",    text_color=color.white, text_size=size.small)
    table.cell(dash, 1, 4, str.tostring(math.round(stochK, 1)),
         text_color=stochK < 20 ? color.lime : stochK > 80 ? color.red : color.silver, text_size=size.small)

    table.cell(dash, 0, 5, "MACD Hist",   text_color=color.white, text_size=size.small)
    table.cell(dash, 1, 5, hist > hist[1] ? "Rising ↑" : "Falling ↓",
         text_color=hist > hist[1] ? color.lime : color.red, text_size=size.small)

    table.cell(dash, 0, 6, "Volume",      text_color=color.white, text_size=size.small)
    table.cell(dash, 1, 6, highVol ? "High ✓" : "Normal",
         text_color=highVol ? color.lime : color.silver, text_size=size.small)

    table.cell(dash, 0, 7, "Buy Score",   text_color=color.white, text_size=size.small)
    table.cell(dash, 1, 7, str.tostring(buyScore) + "/6",
         text_color=buyScore >= minPoints ? color.lime : color.silver, text_size=size.small)

    table.cell(dash, 0, 8, "Sell Score",  text_color=color.white, text_size=size.small)
    table.cell(dash, 1, 8, str.tostring(sellScore) + "/6",
         text_color=sellScore >= minPoints ? color.red : color.silver, text_size=size.small)

// ══════════════════════════════════════════════
// BAR COLOR & ALERTS
// ══════════════════════════════════════════════
barcolor(buySignal ? color.new(color.lime, 60) : sellSignal ? color.new(color.red, 60) : na)

alertcondition(buySignal,  "BUY Signal",  "Confluence BUY — Score signaled")
alertcondition(sellSignal, "SELL Signal", "Confluence SELL — Score signaled")
alertcondition(buySignal  and buyGrade  == "A", "Grade A BUY",  "High-conviction BUY signal")
alertcondition(sellSignal and sellGrade == "A", "Grade A SELL", "High-conviction SELL signal")
